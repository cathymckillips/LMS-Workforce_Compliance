--SQLQuery - Create a stored procedure to run LoadWeeklyComplianceSnapshot-
CREATE PROCEDURE dbo.usp_LoadWeeklyComplianceSnapshot
AS
BEGIN

    SET NOCOUNT ON;

    DECLARE @ReportDate DATE = CAST(GETDATE() AS date);

    IF NOT EXISTS
    (
        SELECT 1
        FROM dbo.Weekly_Compliance_History
        WHERE Report_Date = @ReportDate
    )
    BEGIN

        INSERT INTO dbo.Weekly_Compliance_History
        (
            Report_Date,
            Department,
            Location,
            Employee_Count,
            Total_Assignments,
            Completed_Count,
            Overdue_Count,
            In_Progress_Count,
            Not_Started_Count,
            Due_Next_30_Days,
            Average_Score,
            Compliance_Rate
        )

        SELECT
            Report_Date,
            Department,
            Location,
            Employee_Count,
            Total_Assignments,
            Completed_Count,
            Overdue_Count,
            In_Progress_Count,
            Not_Started_Count,
            Due_Next_30_Days,
            Average_Score,
            Compliance_Rate

        FROM dbo.vw_WeeklyCompliance;

    END;

END;
